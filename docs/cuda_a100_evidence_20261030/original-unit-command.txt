set -euo pipefail
MEFT_GPU_WORK=/llm-align/duyimin/meft-cuda-20261030
cd "$MEFT_GPU_WORK/source"
export CUDA_VISIBLE_DEVICES=3
export OMP_NUM_THREADS=1
export USE_TF=0
export MEFT_TINYENGRAM_SOURCE="$MEFT_GPU_WORK/upstream"
"$MEFT_GPU_WORK/compat/bin/python" -u - <<'PY' > "$MEFT_GPU_WORK/results/cuda-unit-tests.txt" 2>&1
import json, subprocess, sys, torch, unittest
assert torch.cuda.is_available(), 'CUDA must actually be available'
torch.set_num_threads(1)
torch.set_num_interop_threads(1)
free,total=torch.cuda.mem_get_info()
if free < 4*1024**3:
    raise SystemExit('Not enough free GPU memory for conservative shared tests')
torch.cuda.set_per_process_memory_fraction(0.01,0)
print(json.dumps({'python':sys.version,'torch':torch.__version__,'cuda':torch.version.cuda,
                  'device':torch.cuda.get_device_name(),'physical_gpu':3,
                  'free_before':free,'total':total,'allocator_fraction_limit':0.01},indent=2),flush=True)
result=unittest.main(module=None,argv=['unittest','tests.test_device_row_adapter',
                                      'tests.test_tinyengram_checkpoint','-v'],exit=False).result
torch.cuda.synchronize()
print(json.dumps({'max_allocated':torch.cuda.max_memory_allocated(),
                  'max_reserved':torch.cuda.max_memory_reserved(),
                  'free_after':torch.cuda.mem_get_info()[0]},indent=2),flush=True)
raise SystemExit(0 if result.wasSuccessful() else 1)
PY
cat "$MEFT_GPU_WORK/results/cuda-unit-tests.txt"
